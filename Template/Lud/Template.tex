  ll binpowmod(ll a, ll b, ll m){
    a \%= m;
    ll res = 1;
    while ( b > 0 ){
      if ( b & 1 ) res = res * a \% m;
      a = a * a \% m;
      b >>= 1;
    }

void swap_two_iters(vector<int> vec, int n){
